TYPE 5: BACKUP SYNC - COMPLETE COMMAND SEQUENCE
================================================================================

# Step 1: Run rsync to backup
# ───────────────────────────
rsync -av --checksum $AZAZAEL_SOURCE/ $AZAZAEL_BACKUP/

# Step 2: Verify integrity
# ───────────────────────
# Compare file counts
SOURCE_COUNT=$(find $AZAZAEL_SOURCE -type f | wc -l)
BACKUP_COUNT=$(find $AZAZAEL_BACKUP -type f | wc -l)
echo "Source files: $SOURCE_COUNT"
echo "Backup files: $BACKUP_COUNT"

# Compare checksums
diff <(find $AZAZAEL_SOURCE -type f -exec md5sum {} \; | sort) \
     <(find $AZAZAEL_BACKUP -type f -exec md5sum {} \; | sort) \
     && echo "CHECKSUMS MATCH" || echo "CHECKSUMS DIFFER"

# Step 3: Get backup size
# ─────────────────────
BACKUP_SIZE=$(du -sh $AZAZAEL_BACKUP | cut -f1)
echo "Backup size: $BACKUP_SIZE"

# Step 4: Update update_log.txt
# ────────────────────────────

================================================================================

Integrity: VERIFIED
