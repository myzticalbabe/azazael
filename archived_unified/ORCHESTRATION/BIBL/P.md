@staticmethod
def detect_framework(model) -> Optional[str]:
    """Detect model framework from type."""
    
    model_type = type(model).__module__
    
    # Check PyTorch
    if hasattr(model, "state_dict"):
        return "pytorch"
    
    # Check TensorFlow/Keras
    if hasattr(model, "variables") or hasattr(model, "get_config"):
        return "tensorflow"
    
    # Check sklearn
    if "sklearn" in model_type:
        return "sklearn"
    
    # Check XGBoost
    if "xgboost" in model_type:
        return "xgboost"
    
    # Check LightGBM
    if "lightgbm" in model_type:
        return "lightgbm"
    
    return None

    """Auto-detect model framework from type."""

    framework_map = {
        "sklearn": ["sklearn", "sklearn"],
        "torch": ["torch", "pytorch"],
        "tensorflow": ["tensorflow", "tf"],
        "keras": ["keras"],
        "xgboost": ["xgboost"],
        "lightgbm": ["lightgbm"],
        "catboost": ["catboost"],
    }
    
    for framework, prefixes in framework_map.items():
        if any(model_type.startswith(p) for p in prefixes):
            return framework
    
    return "generic"
